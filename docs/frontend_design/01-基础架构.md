### 3.2 权限守卫（beforeEach）

```ts
// router/index.ts
router.beforeEach(async (to) => {
  const userStore = useUserStore()

  // ① 免登录页：已登录访问 /login → 回首页
  if (to.meta.public) {
    if (userStore.token) return { path: '/home' }
    return true
  }

  // ② 未登录 → 登录页（带 redirect 回跳）
  if (!userStore.token) {
    return { path: '/login', query: { redirect: to.fullPath } }
  }

  // ③ 有 token 但用户信息未加载（刷新页面场景）→ 拉取 /me
  if (!userStore.user) {
    try {
      await userStore.fetchMe()          // /me 失败(401)内部会清 token 并抛错
    } catch {
      return { path: '/login', query: { redirect: to.fullPath } }
    }
  }

  // ④ 角色校验：meta.roles 不含当前角色 → 回首页 + 提示
  if (to.meta.roles && !to.meta.roles.includes(userStore.user.role_id)) {
    ElMessage.warning('无权访问该页面')
    return { path: '/home' }
  }

  document.title = `${to.meta.title} · 蜀道安全助手`
  return true
})
```

## 4. Axios 封装（src/api/request.ts）

### 4.1 单例与拦截器

```ts
import axios, { AxiosError, type AxiosInstance, type AxiosResponse } from 'axios'
import { ElMessage } from 'element-plus'
import { useUserStore } from '@/store/user'
import type { ApiResponse } from '@/types/api'

const BASE_URL = '/api/v1'          // 相对路径：dev 走 Vite 代理，prod 走 Nginx 同源

function createInstance(): AxiosInstance {
  const instance = axios.create({ baseURL: BASE_URL, timeout: 15000 })

  // 请求拦截：注入 Bearer token（从 userStore/localStorage 读取）
  instance.interceptors.request.use((config) => {
    const token = localStorage.getItem('shudao_token')
    if (token) config.headers.Authorization = `Bearer ${token}`
    return config
  })

  instance.interceptors.response.use(
    (res) => res,                    // 解包逻辑在调用方（request/uploadRequest）分别处理
    (err: AxiosError) => {
      const status = err.response?.status
      if (status === 401) onUnauthorized()
      return Promise.reject(err)
    },
  )
  return instance
}

// 401 统一处理：清 token → 跳登录（防重复跳转用标记位）
let _redirecting = false
function onUnauthorized() {
  const userStore = useUserStore()
  userStore.clear()
  if (!_redirecting) {
    _redirecting = true
    ElMessage.error('登录已过期，请重新登录')
    const current = router.currentRoute.value.fullPath
    router.replace({ path: '/login', query: { redirect: current } }).finally(() => {
      _redirecting = false
    })
  }
}
```

### 4.2 标准请求：JSON 解包（`request`）

```ts
export async function request<T = unknown>(
  config: AxiosRequestConfig,
): Promise<T> {
  try {
    const res: AxiosResponse<ApiResponse<T>> = await instance.request(config)
    const body = res.data
    // 统一响应 {code, message, data}；code!==200 视为业务失败
    if (body.code !== 200) {
      ElMessage.error(body.message || '请求失败')
      throw new ApiError(body.code, body.message, body.data)
    }
    return body.data as T
  } catch (e) {
    // HTTP 层错误：统一提示（403/404/502/网络）
    handleHttpError(e)
    throw e
  }
}
```

错误提示口径（与后端 main.py 全局异常处理器对齐）：

| HTTP/业务 code | 后端来源 | 前端提示 |
| --- | --- | --- |
| 200 | 业务成功 | 无 |
| 400 | 业务失败（参数/敏感词） | `body.message`（如「内容包含敏感词：xxx」） |
| 401 | 未登录/登录过期 | 清 token + 跳 /login（见 §4.1） |
| 403 | 越权 | 「无权限访问该接口」 |
| 404 | 不存在**或无权**（后端 404 掩码） | 统一「请求的资源不存在」——**不区分**无权/不存在 |
| 422 | Pydantic 校验失败 | `body.message`（如 `page: Input should be...`） |
| 502 | AI 生成/LLM 失败 | 「AI 服务异常，请稍后重试」 |
| 503 | 视觉识别不可用 | 「视觉识别不可用，可跳过或稍后重试」 |
| 网络错误/超时 | 后端不可达 | 「网络异常，请检查后端服务」 |

### 4.3 上传场景（`uploadRequest`）

后端 `/hazards/upload`、`/hazards/analyze` 虽同样返回 `{code,message,data}`，但请求体是 **multipart/FormData**，且业务失败需要**原样拿到 body**（含 url 字段与 503 降级信息）。单独导出、**不做 JSON 解包**：

```ts
export async function uploadRequest<T = unknown>(
  config: AxiosRequestConfig,
): Promise<ApiResponse<T>> {
  try {
    const res = await instance.request<ApiResponse<T>>(config)
    const body = res.data
    // 由调用方判断 code；HTTP 401/网络错误仍走公共拦截器
    return body
  } catch (e) {
    handleHttpError(e)   // 403/404/502/503 统一提示
    throw e
  }
}
```

用法（模块一章节会复用）：

```ts
// api/hazard.ts
export function uploadImage(file: File) {
  const fd = new FormData()
  fd.append('file', file)
  return uploadRequest<{ url: string }>({ url: '/hazards/upload', method: 'post', data: fd })
}
```

> `instance`（baseURL `/api/v1` + Bearer 注入 + 401 处理）是 `request`/`uploadRequest` 共享的**同一个**实例，仅响应解包策略不同。上传接口 URL 用相对 `/hazards/upload`（自动拼接 `/api/v1`）。

### 4.4 前端上传预校验（utils/validate.ts）

发请求前先本地校验，避免无谓流量（PRD B03）：

```ts
export function validateImage(file: File): string | null {
  const allowedTypes = ['image/jpeg', 'image/png']
  const allowedExts = ['.jpg', '.jpeg', '.png']
  const maxBytes = 5 * 1024 * 1024  // ≤5MB
  if (!allowedTypes.includes(file.type)) return '仅支持 jpg/png/jpeg 图片'
  const ext = '.' + (file.name.split('.').pop() ?? '').toLowerCase()
  if (!allowedExts.includes(ext)) return '仅支持 jpg/png/jpeg 图片'
  if (file.size > maxBytes) return '图片不能超过 5MB'
  return null
}
```

## 5. SSE 客户端（src/utils/sse.ts）

### 5.1 为什么用 fetch stream

`EventSource` 无法携带 `Authorization` 头，而 `/api/v1/ai/chat` 需要 Bearer。故用 **fetch + ReadableStream** 读取 body，手动按 SSE 帧切分。后端 `chat.py` 事件序（`qa_service.py` 契约）：`meta → delta* → done`，保活 `ping`。

### 5.2 分帧实现

```ts
export type ChatSSECallback = {
  onMeta:   (d: MetaEvent) => void
  onDelta:  (d: DeltaEvent) => void
  onDone:   (d: DoneEvent) => void
  onError:  (err: Error) => void
}

interface MetaEvent  { conversation_id: number; user_message_id: number; mode: 'full'|'conservative'|'refuse'; confidence: number; citations: Citation[]; rewritten_used: boolean }
interface DeltaEvent { text: string }
interface DoneEvent  { answer_id: number; citations: Citation[]; grounding_score: number; synthetic: boolean; error?: string }

export async function chatSSE(body: object, cb: ChatSSECallback, signal?: AbortSignal) {
  const token = localStorage.getItem('shudao_token')
  const res = await fetch('/api/v1/ai/chat', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json', ...(token ? { Authorization: `Bearer ${token}` } : {}) },
    body: JSON.stringify(body),
    signal,
  })

  // 业务错误在流开始前以 HTTP 状态码返回（敏感词 400 / 未登录 401 / 会话不存在 404）
  if (!res.ok) {
    let message = `请求失败(${res.status})`
    try { message = (await res.json()).message ?? message } catch { /* 忽略 */ }
    cb.onError(new Error(message))
    return
  }

  const reader = res.body!.getReader()
  const decoder = new TextDecoder()
  let buffer = ''

  while (true) {
    const { done, value } = await reader.read()
    if (done) break
    buffer += decoder.decode(value, { stream: true })
    // 按 "\n\n" 分帧；不足一帧留 buffer 等下一块
    let idx: number
    while ((idx = buffer.indexOf('\n\n')) >= 0) {
      const frame = buffer.slice(0, idx)
      buffer = buffer.slice(idx + 2)
      dispatchFrame(frame, cb)
    }
  }
}

function dispatchFrame(frame: string, cb: ChatSSECallback) {
  // 解析 "event: X\ndata: {...}\n\n"
  let event = ''
  const dataLines: string[] = []
  for (const line of frame.split('\n')) {
    if (line.startsWith('event:')) event = line.slice(6).trim()
    else if (line.startsWith('data:')) dataLines.push(line.slice(5).trim())
  }
  if (event === 'ping') return                       // 保活帧：忽略
  if (!dataLines.length) return
  let payload: any
  try { payload = JSON.parse(dataLines.join('\n')) } catch { return } // 脏帧丢弃
  if (event === 'meta')  cb.onMeta(payload)
  else if (event === 'delta') cb.onDelta(payload)
  else if (event === 'done') cb.onDone(payload)
}
```

### 5.3 边界处理

| 场景 | 处理 |
| --- | --- |
| 断连/取消 | 调用方传入 `AbortSignal`；catch `AbortError` 后静默结束，历史已由后端先落库（`qa_service` 先落 user 行） |
| 流中断（网络抖断） | `onError` 触发 → 界面提示「回答已中断」+ 提供重试按钮（重试需带原 conversation_id） |
| `onDone.error` 字段存在 | AI 生成失败：展示「回答生成失败，请重试」 |
| 服务器静默保活 `ping` | 忽略，不刷新 UI |
| 流式期间用户离开页面 | `onBeforeUnmount` 中 `abortController.abort()` |

## 7. Pinia stores

### 7.1 userStore（src/store/user.ts）

```ts
export const useUserStore = defineStore('user', () => {
  const TOKEN_KEY = 'shudao_token'
  const token = ref(localStorage.getItem(TOKEN_KEY) ?? '')
  const user = ref<UserInfo | null>(null)          // {id, username, name, role_id, phone}

  const isLogin = computed(() => !!token.value)

  async function login(username: string, password: string) {
    // POST /auth/login（OAuth2 表单）→ {access_token, token_type, user}
    const data = await request<LoginResp>({
      url: '/auth/login',
      method: 'post',
      headers: { 'Content-Type': 'application/x-www-form-urlencoded' },  // OAuth2 表单格式
      data: new URLSearchParams({ username, password }),
    })
    token.value = data.access_token
    localStorage.setItem(TOKEN_KEY, data.access_token)
    user.value = data.user
    return data.user
  }

  async function fetchMe() {
    const u = await request<UserInfo>({ url: '/auth/me', method: 'get' })
    user.value = u
    return u
  }

  function clear() {
    token.value = ''
    user.value = null
    localStorage.removeItem(TOKEN_KEY)
  }

  function logout() {
    clear()
    router.replace('/login')
  }

  return { token, user, isLogin, login, fetchMe, clear, logout }
})
```

> 登录接口是 **OAuth2 密码模式**：body 必须是 `application/x-www-form-urlencoded`（`username/password`），不能用 JSON。登录响应直接携带 `user`（含 role_id），`fetchMe` 用于刷新页面后恢复登录态（守卫在无 user 时调用）。

### 7.2 appStore（src/store/app.ts）

```ts
export const useAppStore = defineStore('app', () => {
  const collapsed = ref(false)
  const activeMenu = ref('/home')          // 侧栏高亮（由路由变化时更新）
  const breadcrumbs = ref<{ title: string; path: string }[]>([])

  function syncFromRoute(route: RouteLocationNormalizedLoaded) {
    activeMenu.value = route.path
    breadcrumbs.value = route.matched
      .filter(m => m.meta?.title)
      .map(m => ({ title: m.meta.title as string, path: m.path }))
  }
  return { collapsed, activeMenu, breadcrumbs, syncFromRoute }
})
```

在 `App.vue` 或布局 `watch(route)` 调用 `syncFromRoute`。

### 7.3 模块级 store 预留

`store/modules/chat.ts`、`hazard.ts`、`exam.ts` 由对应模块章节定义。约定：**模块级 UI/临时状态放模块 store，跨模块共享的登录态/路由态只放 userStore/appStore**。

## 8. 安全与性能

### 8.1 token 安全

- token 持久化 `localStorage`（key `shudao_token`）；后端 `jwt_expire_minutes=120`（2 小时），到期后任何接口返回 401 → §4.1 统一清 token + 跳登录。
- 刷新页面后由守卫在首次导航时 `fetchMe()` 恢复登录态；`fetchMe` 失败（401）即按过期处理。
- 生产部署建议 HTTPS；本项目实训环境 dev 走本地 http。

### 8.2 XSS 防护

- AI 回答为 Markdown，需 `markdown-it` 渲染并**净化**（`markdown-it` + 手动白名单过滤 `onclick`/`javascript:`/外链协议）。净化细节与可折叠引用在**模块二章节**定义；本页仅约定：**任何 AI 输出不得以 v-html 直接插入，必须经 `utils/markdown.ts` 的 sanitize 管线**。
- 普通文本（隐患描述、题目内容）用 Vue 默认插值（`{{ }}`）自动转义，不得 v-html。
- 后端侧富文本过滤（PRD §6.2）与本页无关，双端各自把关。

### 8.3 性能

| 措施 | 实现 |
| --- | --- |
| 路由懒加载 + 代码分割 | 全部页面 `() => import(...)`；`element-plus` 独立 chunk（§1.4） |
| 首屏 ≤2s | 登录页为最小页面；Layout 懒加载；AI/考试等重页面按需进入 |
| 列表大数据 | 服务端分页（page/page_size），前端不拉全量；`el-table` 用 `v-loading` |
| 图片资源 | `/uploads` URL 由后端生成日期子目录；详情页缩略图用 `el-image` 懒加载 |
| 静态资源 | Vite build 自带 hash + gzip（生产 Nginx 开 gzip） |

### 8.4 上传

前端预校验（§4.4）→ 后端二次校验（`_ALLOWED_EXTS`/`_ALLOWED_MIMES`/`max_upload_mb`）双保险；上传中禁用按钮防重复提交；失败时 `el-upload` 错误回显并允许重传。

## 9. 与后端契约总览（字段级）

| 接口 | 方法 & 路径 | 请求 | 响应 `data` | 权限 |
| --- | --- | --- | --- | --- |
| 登录 | POST `/api/v1/auth/login` | OAuth2 表单 `username/password` | `{access_token, token_type, user:{id,username,name,role_id,phone}}` | 公开 |
| 注册 | POST `/api/v1/auth/register` | JSON `{username,password,name,phone?}` | user 对象 | 公开 |
| 当前用户 | GET `/api/v1/auth/me` | — | `{id,username,name,role_id,phone}` | 登录 |
| AI 问答 | POST `/api/v1/ai/chat` | JSON `{message, conversation_id?}` | **SSE**：`meta→delta*→done` | 登录 |
| 图片上传 | POST `/api/v1/hazards/upload` | multipart `file` | `{url:"/uploads/YYYYMMDD/xxx.jpg"}` | 登录 |
| 图片识别 | POST `/api/v1/hazards/analyze` | multipart `file` | `{url,type_suggest,level_suggest,description,reason,confidence,bbox,detections,annotated_url,report}` | 登录 |

**统一响应**（`app/utils/response.py`）：`{code, message, data}`；业务成功 `code=200`，否则 `code=HTTP 状态码`。前端规则见 §4.2 表。

**SSE 事件字段**（`qa_service._sse`，frames 格式 `event: X\ndata: {...}\n\n`）：

| 事件 | data 字段 | 说明 |
| --- | --- | --- |
| `meta` | `conversation_id, user_message_id, mode(full/conservative/refuse), confidence, citations[], rewritten_used` | 流开始，前端可先渲染引用 |
| `delta` | `text` | 增量文本，逐字追加 |
| `ping` | `{type:"ping"}` | 保活，忽略 |
| `done` | `answer_id, citations[], grounding_score, synthetic, error?` | 结束；`error` 存在=生成失败 |

**后端 URL 常量**（跨章节复用）：后端地址 `http://127.0.0.1:8000`；API 前缀 `/api/v1`；静态上传 `/uploads`。开发期经 Vite 代理同源访问，无需写死后端域名。
